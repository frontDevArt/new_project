```python
def last_market_moments(self, listing_ids) -> dict[str, datetime]:
    ids = list(dict.fromkeys(listing_ids))
    moments: dict[str, datetime] = {}
    for chunk in range(0, len(ids), 500):
        part = ids[chunk:chunk + 500]
        marks = ", ".join("?" * len(part))
        rows = self.conn.execute(
            f"SELECT l.id AS id, "
            f"       MAX(COALESCE((SELECT MAX(p.seen_at) FROM price_history p "
            f"                      WHERE p.listing_id = l.id), l.first_seen), "
            f"           COALESCE(l.returned_at, '')) AS moment "
            f"  FROM listings l WHERE l.id IN ({marks})", part)
        for row in rows:
            if row["moment"]:
                moments[row["id"]] = from_iso(row["moment"])
    return moments
```
